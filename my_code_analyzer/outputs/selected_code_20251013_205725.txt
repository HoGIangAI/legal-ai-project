# File này gộp cả Deployment và Service cho Loki
# Phiên bản này đã được sửa lỗi không khớp tên file cấu hình

apiVersion: apps/v1
kind: Deployment
metadata:
  name: loki-deployment
  labels:
    app: loki
spec:
  replicas: 1
  selector:
    matchLabels:
      app: loki
  template:
    metadata:
      labels:
        app: loki
    spec:
      containers:
        - name: loki
          image: grafana/loki:2.7.0
          args:
            # SỬA LỖI: Trỏ đến đúng tên file cấu hình là loki.yaml
            - "-config.file=/etc/loki/loki.yaml"
          ports:
            - containerPort: 3100
          volumeMounts:
            - name: loki-config-volume
              mountPath: /etc/loki
      volumes:
        - name: loki-config-volume
          configMap:
            name: loki-config

---
apiVersion: v1
kind: Service
metadata:
  name: loki-service
spec:
  selector:
    app: loki
  ports:
    - protocol: TCP
      port: 3100
      targetPort: 3100

apiVersion: traefik.containo.us/v1alpha1
kind: Middleware
metadata:
  name: jwt-authenticator
  namespace: default
spec:
  # Cấu hình này giả định bạn đang sử dụng một plugin tên là 'jwtauth' cho Traefik
  # để xác thực JWT. Cấu hình thực tế có thể thay đổi tùy theo plugin bạn chọn.
  plugin:
    jwtauth:
      # URL để Traefik lấy public key và xác thực chữ ký của JWT token.
      jwksUrl: "http://keycloak-service:8080/realms/master/protocol/openid-connect/certs"
      
      # (Tùy chọn) Chuyển tiếp một số thông tin từ token vào header cho dịch vụ backend.
      forwardHeaders:
        X-Forwarded-User: "preferred_username"
        X-Forwarded-Email: "email"
