  name: intent-system-deployment

    app: intent-system

  replicas: 2
  selector:
    matchLabels:
      app: intent-system
  template:
    metadata:
      labels:
        app: intent-system
      # Annotations để Prometheus có thể tự động khám phá
      annotations:
        prometheus.io/scrape: 'true'
        prometheus.io/path: '/metrics'
        prometheus.io/port: '8001'
    spec:
      containers:
      - name: intent-system
        image: busybox:1.36
        imagePullPolicy: IfNotPresent
        ports:
        - containerPort: 8001
        # --- CẬP NHẬT: THÊM CÁC BIẾN MÔI TRƯỜNG ---
        env:
        - name: KAFKA_BOOTSTRAP_SERVERS
          value: "kafka-service:29092"
        - name: POSTGRES_HOST
          value: "postgres-service" # Giả sử tên service của Postgres là postgres-service
        - name: POSTGRES_DB
          value: "legal_rag"
        - name: POSTGRES_USER
          valueFrom:
            secretKeyRef:
              name: postgres-secret
              key: username
        - name: POSTGRES_PASSWORD
          valueFrom:
            secretKeyRef:
              name: postgres-secret
              key: password
        # --- KẾT THÚC CẬP NHẬT ---
        # Cấu hình tài nguyên
        resources:
          requests:
            cpu: "100m"
            memory: "128Mi"
          limits:
            cpu: "500m"
            memory: "256Mi"
        # Health probes để K8s tự động kiểm tra sức khỏe
        livenessProbe:
          httpGet:
            path: /health
            port: 8001
          initialDelaySeconds: 15
          periodSeconds: 20

          httpGet:
            path: /health
            port: 8001

apiVersion: apps/v1
kind: Deployment
metadata:
  name: prometheus-deployment
spec:
  replicas: 1
  selector:
    matchLabels:
      app: prometheus
  template:
    metadata:
      labels:
        app: prometheus
    spec:
      # Gán ServiceAccount đã tạo ở bước RBAC
      serviceAccountName: prometheus
      containers:
        - name: prometheus
          image: prom/prometheus:v2.37.0
          args:
            - "--config.file=/etc/prometheus/prometheus.yml"
          ports:
            - containerPort: 9090
          # Mount file cấu hình từ ConfigMap vào trong container
          volumeMounts:
            - name: prometheus-config-volume
              mountPath: /etc/prometheus/
      volumes:
        - name: prometheus-config-volume
          configMap:
            name: prometheus-config
