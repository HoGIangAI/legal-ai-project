# k8s/cicd-rbac.yaml
# File này định nghĩa Service Account và các quyền hạn giới hạn cho CI/CD pipeline.

apiVersion: v1
kind: ServiceAccount
metadata:
  name: cicd-service-account
  namespace: default

---
apiVersion: rbac.authorization.k8s.io/v1
kind: Role
metadata:
  name: deployment-manager-role
  namespace: default
rules:
  # Cấp quyền quản lý các tài nguyên Deployments và Services
  - apiGroups: ["apps", ""]
    resources: ["deployments", "services"]
    verbs: ["create", "get", "list", "update", "patch", "delete"]

---
apiVersion: rbac.authorization.k8s.io/v1
kind: RoleBinding
metadata:
  name: cicd-role-binding
  namespace: default
subjects:
  # Gắn các quyền hạn ở trên vào Service Account cicd-service-account
  - kind: ServiceAccount
    name: cicd-service-account
    namespace: default
roleRef:
  kind: Role
  name: deployment-manager-role
  apiGroup: rbac.authorization.k8s.io

apiVersion: v1
kind: ConfigMap
metadata:
  name: prometheus-config
data:
  # Tên file cấu hình bên trong ConfigMap
  prometheus.yml: |
    global:
      scrape_interval: 15s # Thu thập metrics mỗi 15 giây

    scrape_configs:
      # Job để Prometheus tự thu thập metrics của chính nó
      - job_name: 'prometheus'
        static_configs:
          - targets: ['localhost:9090']

      # Job để tự động khám phá và thu thập metrics từ các pod trong Kubernetes
      - job_name: 'kubernetes-pods'
        kubernetes_sd_configs:
          - role: pod
        # relabel_configs dùng để lọc và tùy chỉnh các target được khám phá
        relabel_configs:
          # Chỉ giữ lại các pod có annotation prometheus.io/scrape = true
          - source_labels: [__meta_kubernetes_pod_annotation_prometheus_io_scrape]
            action: keep
            regex: true
          # Lấy path từ annotation để scrape (mặc định là /metrics)
          - source_labels: [__meta_kubernetes_pod_annotation_prometheus_io_path]
            action: replace
            target_label: __metrics_path__
            regex: (.+)
          # Lấy port từ annotation
          - source_labels: [__address__, __meta_kubernetes_pod_annotation_prometheus_io_port]
            action: replace
            regex: ([^:]+)(?::\d+)?;(\d+)
            replacement: $1:$2
            target_label: __address__
          # Gán nhãn cho metrics bằng tên namespace của pod
          - source_labels: [__meta_kubernetes_namespace]
            action: replace
            target_label: kubernetes_namespace
          # Gán nhãn cho metrics bằng tên pod
          - source_labels: [__meta_kubernetes_pod_name]
            action: replace
            target_label: kubernetes_pod_name
